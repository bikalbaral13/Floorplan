_nb = rows(_BX)
_N = 4 * _kA
_jT = arange(_N)
_cc = floor(_jT / _kA)
_kk = _jT - _cc * _kA
_ang = (_cc - 1) * PI / 2 + (_kA > 1 ? (_kk / (_kA - 1)) : 0.5) * PI / 2
_oN = ones(_N, 1)
_oB = ones(_nb, 1)
_X0 = kron(col(_BX, 0), _oN)
_X1 = kron(col(_BX, 1), _oN)
_Y0 = kron(col(_BX, 2), _oN)
_Y1 = kron(col(_BX, 3), _oN)
_E0 = kron(col(_BX, 4), _oN)
_E1 = kron(col(_BX, 5), _oN)
_Rr = kron(col(_BX, 6), _oN)
_Rr = max(0, min(_Rr, 0.499 * min(_X1 - _X0, _Y1 - _Y0)))
_Ax = kron(col(_BX, 7), _oN)
_Gp = kron(col(_BX, 8), _oN)
_Bi = kron(arange(_nb), _oN)
_Cc = kron(_oB, _cc)
_An = kron(_oB, _ang)
_Jt = kron(_oB, _jT)
_isRt = le(_Cc, 1)
_isTp = eq(_Cc, 1) + eq(_Cc, 2)
_pa = where(_isRt, _X1 - _Rr, _X0 + _Rr) + _Rr * cos(_An)
_pb = where(_isTp, _Y1 - _Rr, _Y0 + _Rr) + _Rr * sin(_An)
_last = eq(_Jt, _N - 1)
_na = where(_last, shift(_pa, _N - 1), shift(_pa, -1))
_nbb = where(_last, shift(_pb, _N - 1), shift(_pb, -1))
_M = rows(_pa)
_fid = _Bi * (_N + 2) + _Jt
_SIDE = reshape(hcat(_pa, _pb, _E0, _na, _nbb, _E0, _na, _nbb, _E1, _pa, _pb, _E1), 4 * _M, 3)
_o4r = ones(4, 1)
_sAx = kron(_Ax, _o4r)
_sId = kron(_fid, _o4r)
_sG = kron(_Gp, _o4r)
_TOP = hcat(_pa, _pb, _E1)
_rev = _Bi * _N + (_N - 1 - _Jt)
_BOT = sortby(hcat(_pa, _pb, _E0, _Bi * (_N + 2) + _N + 1, _Ax, _Gp), _rev)
_LAB = vcat(_SIDE, _TOP, sub(_BOT, 0, 0, _M, 3))
_LID = vcat(_sId, _Bi * (_N + 2) + _N, col(_BOT, 3))
_LAX = vcat(_sAx, _Ax, col(_BOT, 4))
_LGP = vcat(_sG, _Gp, col(_BOT, 5))
# (a, b, e) -> (u, v, z): axis 0 is (a, b, e); axis 1 is (e, a, b)
_la = col(_LAB, 0)
_lb = col(_LAB, 1)
_le = col(_LAB, 2)
_isA1 = gt(_LAX, 0.5)
_lu = where(_isA1, _le, _la)
_lv = where(_isA1, _la, _lb)
_lz = where(_isA1, _lb, _le)
# the frame rule, once for every box
_wu = _cu + _lu
_wv = _cv + _lv
_ALL = hcat(_wu * _ux + _wv * _vx, _wu * _uy + _wv * _vy, _lz, _LID, _LGP)
_nAll = rows(_ALL)
